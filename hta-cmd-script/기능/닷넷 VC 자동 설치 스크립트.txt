


@echo off
echo ===== 자동 설치 시작 =====

:: .NET Framework 3.5 설치 (Windows 기능)
DISM /Online /Enable-Feature /FeatureName:NetFx3 /All /Quiet /NoRestart

:: .NET Framework 4.8.1 설치 (다운로드 받은 설치 파일 사용)
start /wait dotNetFx481.exe /quiet /norestart

:: Visual C++ Redistributable 설치 (x64 / x86)
start /wait vcredist_x64.exe /quiet /norestart
start /wait vcredist_x86.exe /quiet /norestart

echo ===== 설치 완료 =====
pause

DISM /Online /Enable-Feature /FeatureName:NetFx3 /All /Quiet /NoRestart
